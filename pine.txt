     minval=0.0,

)

     "Show DD Score",

     "Show Ulcer Index",
     group="5. Display"
)

     "Show Buy/Sell Reasons",
     group="5. Display"

     false,
     "Show Sortino Cross Markers",
     group="5. Display"
)

showNormalRsiCrosses = input.bool(
     false,
     "Show OHLC4 RSI Cross Markers",
     group="5. Display"
)


// =====================================================
// RISK-FREE RATE ADJUSTMENT
// =====================================================
//
// Same method as your supplied Sortino script.
//

minutesPerYear = 525600.0

barsPerYear =
     minutesPerYear /
     timeframe.multiplier

adjustedRiskFreeRate =
     math.pow(
         1.0 + riskFreeRate,
         1.0 / barsPerYear
     ) - 1.0


// =====================================================
// ROLLING SORTINO
// =====================================================
//
// Matches your original implementation:
//
// return
// -> SMA return
// -> STDEV of min(return, 0)
// -> excess return / downside deviation
// -> SMA 14 smoothing
//

currentDailyReturn =
     ta.change(close) /
     close[1]

currentMeanReturn =
     ta.sma(
         currentDailyReturn,
         sortinoLength
     )

currentDownsideReturn =
     math.min(
         currentDailyReturn,
         0.0
     )

currentDownsideDeviation =
     ta.stdev(
         currentDownsideReturn,
         sortinoLength
     )

currentExcessReturn =
     currentMeanReturn -
     adjustedRiskFreeRate

currentSortinoRatio =
     currentDownsideDeviation != 0.0
     ? currentExcessReturn /
       currentDownsideDeviation
     : na


// =====================================================
// SMA 14 OF SORTINO
// =====================================================

smoothedSortino =
     ta.sma(
         currentSortinoRatio,
         sortinoSmoothingLength
     )


// =====================================================
// RSI 14 OF SORTINO
// =====================================================

sortinoRSI =
     ta.rsi(
         smoothedSortino,
         sortinoRsiLength
     )


// =====================================================
// EMA 14 OF SORTINO RSI
// =====================================================

sortinoRsiEMA =
     ta.ema(
         sortinoRSI,
         sortinoRsiEmaLength
     )


// =====================================================
// SORTINO CROSS EVENTS
// =====================================================

sortinoCrossUp =
     barstate.isconfirmed and
     ta.crossover(
         sortinoRSI,
         sortinoRsiEMA
     )

sortinoCrossDown =
     barstate.isconfirmed and
     ta.crossunder(
         sortinoRSI,
         sortinoRsiEMA
     )


// =====================================================
// NORMAL RSI 14 OF OHLC4
// =====================================================
//
// IMPORTANT:
//
// This is:
//
// RSI(14, OHLC4)
//
// NOT RSI of close.
//

normalRSI =
     ta.rsi(
         ohlc4,
         normalRsiLength
     )

normalRsiEMA =
     ta.ema(
         normalRSI,
         normalRsiMaLength
     )


// =====================================================
// OHLC4 RSI CROSS EVENTS
// =====================================================

normalRsiCrossDown =
     barstate.isconfirmed and
     ta.crossunder(
         normalRSI,
         normalRsiEMA
     )

normalRsiCrossUp =
     barstate.isconfirmed and
     ta.crossover(
         normalRSI,
         normalRsiEMA
     )

normalRsiAbove =
     normalRSI >
     normalRsiEMA

normalRsiBelow =
     normalRSI <
     normalRsiEMA


// =====================================================
// DD SCORE FUNCTION
// =====================================================

calcDDScore(src) =>

    ret =
         src / src[1] - 1.0


    // Mean return
    meanReturn =
         ta.ema(
             ret,
             ddReturnLength
         )


    // Downside return
    downsideReturn =
         math.min(
             ret,
             0.0
         )

    downsideSquared =
         downsideReturn *
         downsideReturn


    // Downside deviation
    downsideVariance =
         ta.ema(
             downsideSquared,
             ddDownsideLength
         )

    downsideDeviation =
         math.sqrt(
             downsideVariance
         )


    // Drawdown
    highestClose =
         ta.highest(
             src,
             ddDrawdownLength
         )

    drawdown =
         highestClose > 0
         ? src / highestClose - 1.0
         : 0.0

    absoluteDrawdown =
         math.max(
             -drawdown,
             0.0
         )


    // Drawdown penalty
    drawdownPenalty =
         absoluteDrawdown *
         ddWeight


    // Total downside risk
    totalRisk =
         downsideDeviation +
         drawdownPenalty


    // Raw DD score
    rawScore =
         totalRisk > 0
         ? meanReturn / totalRisk
         : 0.0


    // Smooth DD score
    finalScore =
         ta.ema(
             rawScore,
             ddSmoothLength
         )

    finalScore


// =====================================================
// DD SCORE
// =====================================================

ddScore =
     calcDDScore(
         close
     )

ddPositive =
     ddScore > 0.0

ddNegative =
     ddScore < 0.0


// =====================================================
// ULCER INDEX
// =====================================================

ulcerPeak =
     ta.highest(
         close,
         ulcerPeakLength
     )

ulcerDrawdown =
     ulcerPeak > 0
     ? 100.0 *
       (close / ulcerPeak - 1.0)
     : 0.0

ulcerSquared =
     ulcerDrawdown *
     ulcerDrawdown

ulcerIndex =
     math.sqrt(
         ta.sma(
             ulcerSquared,
             ulcerLength
         )
     )

ulcerHigh =
     ulcerIndex >
     ulcerMinimum


// =====================================================
// BUY CROSS MEMORY
// =====================================================

var int bullCrossBar = na


if sortinoCrossUp
    bullCrossBar := bar_index


// A bearish Sortino cross cancels an unconfirmed buy.
if sortinoCrossDown
    bullCrossBar := na


// Expire old bullish setup.
if not na(bullCrossBar)

    if bar_index - bullCrossBar > confirmationWindow
        bullCrossBar := na


barsAfterBullCross =
     not na(bullCrossBar)
     ? bar_index - bullCrossBar
     : na

validBullCross =
     not na(bullCrossBar) and
     barsAfterBullCross >= 0 and
     barsAfterBullCross <= confirmationWindow

sortinoBullNow =
     sortinoRSI >
     sortinoRsiEMA


// =====================================================
// POSITION STATE
// =====================================================

var bool inPosition = false
var int entryBar = na


// =====================================================
// BEARISH EXIT STATE
// =====================================================
//
// The Sortino bearish cross happens FIRST.
//
// If OHLC4 RSI is already below its EMA,
// RSI confirmation already exists.
//
// If OHLC4 RSI is still ABOVE its EMA,
// wait until OHLC4 RSI crosses DOWN.
//

var bool bearishSortinoWarning = false

var bool waitingForNormalRsi = false

var int bearishSortinoBar = na


// =====================================================
// SORTINO CROSSES DOWN WHILE LONG
// =====================================================

if inPosition and
   sortinoCrossDown

    bearishSortinoWarning := true
    bearishSortinoBar := bar_index

    // Normal OHLC4 RSI is still bullish:
    // WAIT for its bearish crossover.
    if normalRsiAbove
        waitingForNormalRsi := true

    // Normal OHLC4 RSI is already bearish:
    // No need to wait for another crossover.
    else
        waitingForNormalRsi := false


// =====================================================
// OHLC4 RSI LATER CROSSES DOWN
// =====================================================
//
// Only matters after Sortino already warned.
//

if inPosition and
   bearishSortinoWarning and
   waitingForNormalRsi and
   normalRsiCrossDown

    waitingForNormalRsi := false


// =====================================================
// SORTINO RECOVERS BEFORE RSI CONFIRMS
// =====================================================
//
// Cancel the pending normal exit if the Sortino RSI
// crosses back above its EMA before confirmation.
//

if inPosition and
   bearishSortinoWarning and
   sortinoCrossUp

    bearishSortinoWarning := false
    waitingForNormalRsi := false
    bearishSortinoBar := na


// =====================================================
// BUY RULE
// =====================================================
//
// BUY:
//
// 1. RSI14 of smoothed Rolling Sortino14
//    crossed above EMA14.
//
// 2. Cross occurred within previous 10 bars.
//
// 3. Sortino RSI remains above EMA.
//
// 4. DD Score > 0.
//
// 5. Ulcer Index > 10.
//
// Normal OHLC4 RSI is NOT used for entry.
//

buySignal =
     barstate.isconfirmed and
     not inPosition and
     validBullCross and
     sortinoBullNow and
     ddPositive and
     ulcerHigh


// =====================================================
// NORMAL SELL CONFIRMATION
// =====================================================
//
// Sortino crossed DOWN first.
//
// If OHLC4 RSI was above EMA:
//     wait for RSI cross down.
//
// If OHLC4 RSI was already below EMA:
//     no waiting.
//
// DD must be < 0.
//

normalExitReady =
     bearishSortinoWarning and
     not waitingForNormalRsi and
     normalRsiBelow


normalSellSignal =
     barstate.isconfirmed and
     inPosition and
     normalExitReady and
     ddNegative


// =====================================================
// HARD DD >= 3 SELL
// =====================================================
//
// Overrides everything.
//
// No Sortino cross required.
// No OHLC4 RSI cross required.
//

hardDDSell =
     barstate.isconfirmed and
     inPosition and
     not na(entryBar) and
     bar_index > entryBar and
     ddScore >= ddHardSell


// =====================================================
// FINAL SELL
// =====================================================

sellSignal =
     normalSellSignal or
     hardDDSell


// =====================================================
// EXECUTE BUY
// =====================================================

if buySignal

    inPosition := true
    entryBar := bar_index

    bullCrossBar := na

    bearishSortinoWarning := false
    waitingForNormalRsi := false
    bearishSortinoBar := na


// =====================================================
// EXECUTE SELL
// =====================================================

if sellSignal

    inPosition := false
    entryBar := na

    bullCrossBar := na

    bearishSortinoWarning := false
    waitingForNormalRsi := false
    bearishSortinoBar := na
